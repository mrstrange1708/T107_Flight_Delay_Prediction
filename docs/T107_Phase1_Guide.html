"""Phase 2 data collection + labelling + Phase 5 splitting.
Owner: Rishik."""
import subprocess
import sys
import tarfile
import tempfile
import zipfile
from pathlib import Path

import numpy as np
import pandas as pd

from .config import DATA_RAW, DELAY_THRESHOLD_MIN, SPLITS, WARMUP_END


def download_raw(force: bool = False) -> Path:
    """Get the NYC 2013 flights tables (the Kaggle 'Flights' data) into data/raw/.

    The data ship inside the `nycflights13` package source on PyPI. We download the source
    archive with pip and extract the CSVs directly, which avoids the package's broken installer
    on new Python versions. If you already have Kaggle's flights.csv, put it in data/raw/.
    """
    DATA_RAW.mkdir(parents=True, exist_ok=True)
    if (DATA_RAW / "flights.csv").exists() and (DATA_RAW / "planes.csv").exists() and not force:
        return DATA_RAW
    with tempfile.TemporaryDirectory() as tmp:
        subprocess.run([sys.executable, "-m", "pip", "download", "nycflights13==0.0.3", "--no-deps",
                        "--no-binary", ":all:", "-d", tmp, "-q"], check=True)
        archive = next(Path(tmp).glob("nycflights13-*.tar.gz"))
        with tarfile.open(archive) as tar:
            for m in tar.getmembers():
                name = Path(m.name).name
                if "/data/" in m.name and name in {"flights.csv.zip", "planes.csv", "airlines.csv", "airports.csv", "weather.csv"}:
                    with tar.extractfile(m) as src, open(DATA_RAW / name, "wb") as dst:
                        dst.write(src.read())
    with zipfile.ZipFile(DATA_RAW / "flights.csv.zip") as z:
        inner = next(n for n in z.namelist() if n.endswith(".csv") and not n.startswith("__"))
        with z.open(inner) as src, open(DATA_RAW / "flights.csv", "wb") as dst:
            dst.write(src.read())
    (DATA_RAW / "flights.csv.zip").unlink()
    return DATA_RAW


def load_raw() -> tuple[pd.DataFrame, pd.DataFrame]:
    download_raw()
    flights = pd.read_csv(DATA_RAW / "flights.csv")
    flights = flights.drop(columns=[c for c in ("Unnamed: 0", "id", "name") if c in flights.columns])
    planes = pd.read_csv(DATA_RAW / "planes.csv")
    return flights, planes


def hhmm_to_minutes(x: pd.Series) -> pd.Series:
    x = x.astype(int) % 2400
    return (x // 100) * 60 + x % 100


def build_dataset(flights: pd.DataFrame, planes: pd.DataFrame) -> pd.DataFrame:
    """One row per scheduled flight, with the label and static aircraft attributes attached."""
    df = flights.copy()
    df["date"] = pd.to_datetime(df[["year", "month", "day"]])
    df["sched_dep_min"] = hhmm_to_minutes(df["sched_dep_time"])
    df["sched_arr_min"] = hhmm_to_minutes(df["sched_arr_time"])
    df["sched_hour"] = df["sched_dep_min"] // 60
    df["arr_hour"] = df["sched_arr_min"] // 60
    # Cancelled flights never departed; diverted flights departed but never reached the destination.
    df["cancelled"] = df["dep_time"].isna()
    df["diverted"] = df["dep_time"].notna() & df["arr_delay"].isna()
    # Label: 1 if arrival delay >= 15 min, NaN where no arrival delay exists (cancelled/diverted).
    df["delayed"] = np.where(df["arr_delay"].isna(), np.nan, (df["arr_delay"] >= DELAY_THRESHOLD_MIN).astype(float))
    p = planes[["tailnum", "year", "seats", "engines"]].rename(
        columns={"year": "plane_year", "seats": "plane_seats", "engines": "plane_engines"})
    df = df.merge(p, on="tailnum", how="left")
    df["plane_age"] = 2013 - df["plane_year"]
    df["tailnum"] = df["tailnum"].fillna("UNKNOWN")
    return df.sort_values(["date", "sched_dep_min", "origin", "carrier", "flight"]).reset_index(drop=True)


def split(df: pd.DataFrame, labelled_only: bool = True) -> dict:
    """Phase 5: chronological split. Returns {'train','valid','calib','test'} frames."""
    d = df[df["delayed"].notna()] if labelled_only else df
    return {k: d[(d["date"] >= a) & (d["date"] <= b)] for k, (a, b) in SPLITS.items()}

"""Phase 1 success metrics, decided before any model is trained, and used by everyone.
Owner: Rishik."""
import numpy as np
from sklearn.metrics import (average_precision_score, brier_score_loss, log_loss,
                             precision_recall_curve, roc_auc_score)


def ece(y, p, n_bins: int = 15) -> float:
    """Expected calibration error: average |observed rate - mean predicted| over probability bins."""
    y, p = np.asarray(y), np.asarray(p)
    bins = np.minimum((p * n_bins).astype(int), n_bins - 1)
    return float(sum((bins == b).mean() * abs(y[bins == b].mean() - p[bins == b].mean())
                     for b in range(n_bins) if (bins == b).any()))


def at_threshold(y, p, thr: float) -> dict:
    y, p = np.asarray(y).astype(int), np.asarray(p)
    flag = p >= thr
    tp = int((flag & (y == 1)).sum())
    return {"threshold": float(thr), "flag_rate": float(flag.mean()),
            "precision": tp / max(int(flag.sum()), 1), "recall": tp / max(int(y.sum()), 1)}


def threshold_for_precision(y, p, target: float = 0.5) -> float:
    """Lowest threshold whose precision >= target (i.e. the most recall at that precision)."""
    prec, _, thr = precision_recall_curve(y, p)
    ok = np.where(prec[:-1] >= target)[0]
    return float(thr[ok[0]]) if len(ok) else 1.0


def evaluate(y, p) -> dict:
    y = np.asarray(y).astype(int)
    p = np.clip(np.asarray(p, dtype=float), 1e-6, 1 - 1e-6)
    return {"pr_auc": float(average_precision_score(y, p)), "roc_auc": float(roc_auc_score(y, p)),
            "brier": float(brier_score_loss(y, p)), "log_loss": float(log_loss(y, p)), "ece": ece(y, p),
            "base_rate": float(y.mean()), "lift": float(average_precision_score(y, p) / y.mean())}


def day_block_bootstrap(y, p, dates, n_boot: int = 200, seed: int = 0) -> tuple:
    """95% CI of PR AUC by resampling whole DAYS (flights on one day share weather, so they are not independent)."""
    rng = np.random.default_rng(seed)
    y, p = np.asarray(y).astype(int), np.asarray(p)
    _, inv = np.unique(np.asarray(dates), return_inverse=True)
    rows = [np.where(inv == i)[0] for i in range(inv.max() + 1)]
    vals = []
    for _ in range(n_boot):
        idx = np.concatenate([rows[i] for i in rng.integers(0, len(rows), len(rows))])
        vals.append(average_precision_score(y[idx], p[idx]))
    return float(np.percentile(vals, 2.5)), float(np.percentile(vals, 97.5))

"""Phase 4 preprocessing: missing values, scaling, encoding.
Owner: Praneeth.

Rule from the course (Phase 5 warning): every statistic used for imputation or scaling is FIT ON
THE TRAINING ROWS ONLY and then applied to validation/test. Wrapping the steps in a scikit-learn
Pipeline guarantees that, because .fit() only ever sees the rows it is given.
"""
import numpy as np
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import OneHotEncoder, OrdinalEncoder, RobustScaler, StandardScaler


def linear_preprocessor(numeric, categorical, scaler: str = "standard"):
    """For gradient-based / linear models (logistic regression, our GD code).

    numeric:     median imputation + missing-indicator column, then scaling (z-score or robust).
    categorical: one-hot; categories seen < 50 times in training are grouped as 'infrequent',
                 and unseen categories at test time map to that group instead of crashing.
    """
    sc = StandardScaler() if scaler == "standard" else RobustScaler()
    return ColumnTransformer([
        ("num", make_pipeline(SimpleImputer(strategy="median", add_indicator=True), sc), list(numeric)),
        ("cat", OneHotEncoder(handle_unknown="infrequent_if_exist", min_frequency=50), list(categorical)),
    ])


def tree_preprocessor(numeric, categorical):
    """For tree models (decision tree, random forest).

    Trees split on thresholds, so scaling changes nothing and is skipped. Label/ordinal codes are
    safe for trees (no distances are computed). Unseen categories get code -1.
    """
    return ColumnTransformer([
        ("num", SimpleImputer(strategy="median", add_indicator=True), list(numeric)),
        ("cat", OrdinalEncoder(handle_unknown="use_encoded_value", unknown_value=-1), list(categorical)),
    ])


def iqr_outlier_share(x) -> float:
    x = np.asarray(x, dtype=float)
    x = x[~np.isnan(x)]
    q1, q3 = np.percentile(x, [25, 75])
    return float(((x < q1 - 1.5 * (q3 - q1)) | (x > q3 + 1.5 * (q3 - q1))).mean())

Why is aircraft data missing, and what did you do?
Mostly for MQ and AA, whose tail numbers are not in the planes table: systematic. Median imputation plus a missing-indicator, fitted on training rows.
Why keep outliers?

"""Phase 6 feature engineering (creation). Every feature is known at 05:00 on day D.
Owner: Srikar.

Groups
  schedule   : cyclic departure/arrival time, distance, overnight arrival
  calendar   : cyclic day of week, weekend, days to nearest US federal holiday
  aircraft   : plane age, seats, engines (static)
  congestion : how many flights are SCHEDULED at the origin in the hour / +-1 h / day, the carrier's
               flights at the origin that day, NYC arrivals scheduled into the destination that hour
  history    : smoothed delay rate + log volume over trailing windows that END ON DAY D-1
"""
import numpy as np
import pandas as pd
from pandas.tseries.holiday import USFederalHolidayCalendar

# (name, grouping columns, window in days). Windows come from the ACF analysis in notebook 02.
HISTORY_SPECS = [
    ("all", [], 1), ("all", [], 7),
    ("carrier", ["carrier"], 7), ("carrier", ["carrier"], 28),
    ("origin", ["origin"], 1), ("origin", ["origin"], 7),
    ("dest", ["dest"], 28), ("route", ["origin", "dest"], 28),
    ("carrier_origin", ["carrier", "origin"], 7),
    ("origin_hour", ["origin", "sched_hour"], 28),
    ("flight", ["carrier", "flight"], 28),
    ("tail", ["tailnum"], 7),
]
COLD_START_RATE = 0.20   # used only before any history exists (first days of January)


def _key(df, cols):
    if not cols:
        return pd.Series("ALL", index=df.index)
    k = df[cols[0]].astype(str)
    for c in cols[1:]:
        k = k + "|" + df[c].astype(str)
    return k


def trailing_counts(df, cols, window):
    """(delayed count, flight count) per (date, group) over the `window` days BEFORE each date.

    .rolling(window).sum() covers days [D-window+1, D]; .shift(1) moves it to [D-window, D-1],
    so outcomes of day D itself are never included. This one line is what prevents leakage.
    """
    obs = df[df["delayed"].notna()].assign(_g=lambda d: _key(d, cols))
    dates = pd.date_range(df["date"].min(), df["date"].max(), freq="D")
    k = obs.pivot_table(index="date", columns="_g", values="delayed", aggfunc="sum", fill_value=0)
    n = obs.pivot_table(index="date", columns="_g", values="delayed", aggfunc="size", fill_value=0)
    roll = lambda m: m.reindex(dates, fill_value=0).rolling(window, min_periods=1).sum().shift(1).fillna(0)
    k, n = roll(k), roll(n)
    k.index.name = n.index.name = "date"
    return k.stack(), n.stack()


def build_features(df: pd.DataFrame, alpha: float = 10.0):
    """Return (X, groups). X = df plus feature columns; groups = {group name: [columns]}."""
    X = df.copy()
    g = {"schedule": [], "calendar": [], "aircraft": ["plane_age", "plane_seats", "plane_engines"],
         "congestion": [], "history": []}

    # schedule: time of day lives on a circle (23:59 is next to 00:00), so encode with sin/cos
    for col, name in (("sched_dep_min", "dep"), ("sched_arr_min", "arr")):
        ang = 2 * np.pi * X[col] / 1440
        X[f"{name}_sin"], X[f"{name}_cos"] = np.sin(ang), np.cos(ang)
    X["log_distance"] = np.log(X["distance"])
    X["overnight_arrival"] = (X["sched_arr_min"] < X["sched_dep_min"]).astype(int)
    g["schedule"] = ["dep_sin", "dep_cos", "arr_sin", "arr_cos", "sched_hour", "log_distance", "overnight_arrival"]

    # calendar (no month / day-of-year: test months never appear in training, so they act as date IDs)
    dow = X["date"].dt.dayofweek
    X["dow_sin"], X["dow_cos"] = np.sin(2 * np.pi * dow / 7), np.cos(2 * np.pi * dow / 7)
    X["is_weekend"] = (dow >= 5).astype(int)
    hol = USFederalHolidayCalendar().holidays("2012-12-01", "2014-01-31").values
    days = np.abs(X["date"].values[:, None] - hol[None, :]).min(axis=1) / np.timedelta64(1, "D")
    X["days_to_holiday"] = np.minimum(days, 14)
    g["calendar"] = ["dow_sin", "dow_cos", "is_weekend", "days_to_holiday"]

    # congestion from the published schedule (cancelled flights included: they were scheduled)
    per_hour = X.groupby(["origin", "date", "sched_hour"])["flight"].transform("size")
    X["origin_hour_sched"] = per_hour - 1
    oh = X.groupby(["origin", "date", "sched_hour"]).size()
    idx = lambda h: pd.MultiIndex.from_arrays([X["origin"], X["date"], h])
    X["origin_3h_sched"] = sum(oh.reindex(idx(X["sched_hour"] + s)).fillna(0).to_numpy() for s in (-1, 0, 1)) - 1
    X["origin_day_sched"] = X.groupby(["origin", "date"])["flight"].transform("size") - 1
    X["carrier_origin_day_sched"] = X.groupby(["carrier", "origin", "date"])["flight"].transform("size") - 1
    X["dest_arr_hour_sched"] = X.groupby(["dest", "date", "arr_hour"])["flight"].transform("size") - 1
    g["congestion"] = ["origin_hour_sched", "origin_3h_sched", "origin_day_sched", "carrier_origin_day_sched", "dest_arr_hour_sched"]

    # history: smoothed past delay rates. rate = (k + alpha * prior) / (n + alpha)
    priors = {}
    for name, cols, w in HISTORY_SPECS:
        if w not in priors:
            pk, pn = trailing_counts(df, [], w)
            priors[w] = (pk.droplevel(1) / pn.droplevel(1).replace(0, np.nan)).fillna(COLD_START_RATE)
        k, n = trailing_counts(df, cols, w)
        ix = pd.MultiIndex.from_arrays([X["date"], _key(X, cols)])
        kk = k.reindex(ix).fillna(0).to_numpy()
        nn = n.reindex(ix).fillna(0).to_numpy()
        prior = priors[w].reindex(X["date"]).fillna(COLD_START_RATE).to_numpy()
        X[f"h_{name}_{w}d_rate"] = (kk + alpha * prior) / (nn + alpha)
        g["history"].append(f"h_{name}_{w}d_rate")
        if cols:  # volume tells the model how much evidence the rate is based on
            X[f"h_{name}_{w}d_logn"] = np.log1p(nn)
            g["history"].append(f"h_{name}_{w}d_logn")
    return X, g


def numeric_columns(groups, include=("schedule", "calendar", "aircraft", "congestion", "history")):
    return [c for name in include for c in groups[name]]

"""Phase 6 feature selection: filter (correlation / mutual information), wrapper (SFS),
multicollinearity check (VIF), dimensionality reduction check (PCA).
Owner: Srikar."""
import numpy as np
import pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import average_precision_score
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler


def vif(frame: pd.DataFrame) -> pd.Series:
    """VIF_j = 1 / (1 - R^2_j), R^2_j from regressing feature j on all the others (computed via the
    inverse correlation matrix: VIF_j is its j-th diagonal element)."""
    corr = np.corrcoef(frame.to_numpy(dtype=float), rowvar=False)
    return pd.Series(np.diag(np.linalg.pinv(corr)), index=frame.columns)


def vif_prune(frame: pd.DataFrame, limit: float = 10.0):
    """Repeatedly drop the highest-VIF feature until every VIF <= limit."""
    kept, dropped = list(frame.columns), []
    while True:
        v = vif(frame[kept])
        if v.max() <= limit:
            return kept, dropped
        worst = v.idxmax()
        dropped.append((worst, float(v.max())))
        kept.remove(worst)


def lr_score(Xtr, ytr, Xva, yva, cols, C=1.0):
    m = make_pipeline(StandardScaler(), LogisticRegression(C=C, max_iter=2000))
    m.fit(Xtr[cols], ytr)
    return float(average_precision_score(yva, m.predict_proba(Xva[cols])[:, 1]))


def sequential_forward_selection(Xtr, ytr, Xva, yva, candidates, max_features=15, verbose=True):
    """Wrapper method from scratch: start empty, add the feature that most improves validation PR AUC."""
    selected, remaining, path = [], list(candidates), []
    for step in range(max_features):
        scores = {f: lr_score(Xtr, ytr, Xva, yva, selected + [f]) for f in remaining}
        best = max(scores, key=scores.get)
        selected.append(best)
        remaining.remove(best)
        path.append({"step": step + 1, "added": best, "valid_pr_auc": scores[best]})
        if verbose:
            print(f"step {step + 1:2d}: + {best:<28s} valid PR AUC {scores[best]:.4f}")
    return path

"""Phase 7 baselines + Phase 8 model-selection candidates.
Owner: Sami.

Includes logistic regression written from scratch and trained with gradient descent, following
the course's training loop: initialise -> forward pass -> loss -> gradient step -> convergence check.
"""
import numpy as np


def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))


class LogisticGD:
    """Binary logistic regression by gradient descent.

    batch_size=None -> batch GD (all rows per step); 1 -> stochastic GD; k -> mini-batch GD.
    l2 adds (l2/2)*||w||^2 to the loss (ridge penalty; the intercept is not penalised).
    Loss = mean log loss = negative log-likelihood of a Bernoulli label (maximum likelihood).
    Gradient of the loss w.r.t. w = X^T (sigmoid(Xw) - y) / n  (+ l2 * w).
    """

    def __init__(self, lr=0.5, epochs=200, batch_size=None, l2=0.0, tol=1e-6, seed=42):
        self.lr, self.epochs, self.batch_size, self.l2, self.tol, self.seed = lr, epochs, batch_size, l2, tol, seed

    @staticmethod
    def _add_bias(X):
        return np.column_stack([np.ones(len(X)), X])

    def loss(self, A, y):
        p = sigmoid(A @ self.w)
        ll = -np.mean(y * np.log(p + 1e-12) + (1 - y) * np.log(1 - p + 1e-12))
        return float(ll + 0.5 * self.l2 * np.sum(self.w[1:] ** 2))

    def _grad(self, A, y):
        g = A.T @ (sigmoid(A @ self.w) - y) / len(y)
        g[1:] += self.l2 * self.w[1:]
        return g

    def fit(self, X, y):
        rng = np.random.default_rng(self.seed)
        A, y = self._add_bias(np.asarray(X, dtype=float)), np.asarray(y, dtype=float)
        self.w = np.zeros(A.shape[1])                               # 1. initialise
        self.history = [self.loss(A, y)]
        for _ in range(self.epochs):
            if self.batch_size is None:
                self.w -= self.lr * self._grad(A, y)                 # 2-4. forward, loss gradient, step
            else:
                order = rng.permutation(len(y))
                for i in range(0, len(y), self.batch_size):
                    b = order[i:i + self.batch_size]
                    self.w -= self.lr * self._grad(A[b], y[b])
            self.history.append(self.loss(A, y))
            if abs(self.history[-2] - self.history[-1]) < self.tol:  # 5. convergence check
                break
        return self

    def predict_proba(self, X):
        return sigmoid(self._add_bias(np.asarray(X, dtype=float)) @ self.w)

"""Phase 8 gradient boosting: from-scratch implementation, XGBoost tuning with rolling-origin
validation, and Platt calibration.
Owner: Mani Kanta."""
import itertools

import numpy as np
import pandas as pd
import xgboost as xgb
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import average_precision_score
from sklearn.tree import DecisionTreeRegressor


class GradientBoostingScratch:
    """Gradient boosting for binary log loss, written from scratch.

    F_0(x) = log-odds of the training base rate.
    Round m: p = sigmoid(F), negative gradient of log loss r = y - p.
             Fit a small regression tree to r (it groups rows that need similar corrections).
             Leaf value = sum(r) / sum(p(1-p))   (one Newton step for log loss in that leaf).
             F <- F + learning_rate * leaf value.
    """

    def __init__(self, n_estimators=100, learning_rate=0.1, max_depth=3, min_samples_leaf=50, seed=42):
        self.n_estimators, self.lr, self.max_depth, self.min_leaf, self.seed = n_estimators, learning_rate, max_depth, min_samples_leaf, seed

    def fit(self, X, y, X_val=None, y_val=None):
        X, y = np.asarray(X, float), np.asarray(y, float)
        p0 = y.mean()
        self.f0 = np.log(p0 / (1 - p0))
        F = np.full(len(y), self.f0)
        self.trees, self.leaf_values, self.train_loss, self.val_loss = [], [], [], []
        Fv = None if X_val is None else np.full(len(X_val), self.f0)
        for m in range(self.n_estimators):
            p = 1 / (1 + np.exp(-F))
            r = y - p
            tree = DecisionTreeRegressor(max_depth=self.max_depth, min_samples_leaf=self.min_leaf,
                                         random_state=self.seed + m).fit(X, r)
            leaf = tree.apply(X)
            values = {}
            for l in np.unique(leaf):
                mask = leaf == l
                values[l] = r[mask].sum() / max((p[mask] * (1 - p[mask])).sum(), 1e-12)
            F += self.lr * np.vectorize(values.get)(leaf)
            self.trees.append(tree)
            self.leaf_values.append(values)
            self.train_loss.append(self._logloss(y, F))
            if Fv is not None:
                Fv += self.lr * np.vectorize(lambda l: values.get(l, 0.0))(tree.apply(np.asarray(X_val, float)))
                self.val_loss.append(self._logloss(np.asarray(y_val, float), Fv))
        return self

    @staticmethod
    def _logloss(y, F):
        p = np.clip(1 / (1 + np.exp(-F)), 1e-12, 1 - 1e-12)
        return float(-np.mean(y * np.log(p) + (1 - y) * np.log(1 - p)))

    def decision_function(self, X):
        X = np.asarray(X, float)
        F = np.full(len(X), self.f0)
        for tree, values in zip(self.trees, self.leaf_values):
            F += self.lr * np.vectorize(lambda l: values.get(l, 0.0))(tree.apply(X))
        return F

    def predict_proba(self, X):
        return 1 / (1 + np.exp(-self.decision_function(X)))


# ---------------------------------------------------------------- XGBoost
def as_xgb_frame(X: pd.DataFrame, numeric, categorical, categories: dict):
    """Numeric columns as float (NaN allowed: XGBoost learns where missing values go) and
    categoricals as pandas 'category' with a FIXED category list, so codes match across splits."""
    out = X[list(numeric)].astype(float).copy()
    for c in categorical:
        out[c] = pd.Categorical(X[c].astype(str), categories=categories[c])
    return out


def make_xgb(params: dict, n_estimators: int, seed: int = 42, early_stopping: int | None = None):
    kw = dict(objective="binary:logistic", tree_method="hist", enable_categorical=True, max_cat_to_onehot=1,
              n_estimators=n_estimators, random_state=seed, n_jobs=-1, eval_metric="aucpr", **params)
    if early_stopping:
        kw["early_stopping_rounds"] = early_stopping
    return xgb.XGBClassifier(**kw)


def rolling_eval(Xf: pd.DataFrame, y: np.ndarray, month: np.ndarray, params: dict, months=(7, 8, 9),
                 margin_fn=None, early_stopping: int = 100, max_trees: int = 3000, seed: int = 42):
    """Honest rolling-origin score for one configuration.

    For each test month m:
      1. train on months < m-1 and early-stop on month m-1 (picks the number of trees),
      2. refit on ALL months < m with that number of trees,
      3. score month m, which played no part in steps 1-2.
    (Early-stopping on month m and then scoring month m would be optimistic: the stopping rule
    has already seen the answers.)

    margin_fn(train_idx, [idx, ...]) -> [logit arrays]: optional starting prediction for each row
    (e.g. logistic regression's logit). Boosting then learns only corrections on top of it.
    """
    scores, trees = [], []
    for m in months:
        inner, stop = np.where(month < m - 1)[0], np.where(month == m - 1)[0]
        full, test = np.where(month < m)[0], np.where(month == m)[0]
        bi = bs = bf = bt = None
        if margin_fn:
            bi, bs = margin_fn(inner, [inner, stop])
        model = make_xgb(params, max_trees, seed, early_stopping)
        model.fit(Xf.iloc[inner], y[inner], base_margin=bi, eval_set=[(Xf.iloc[stop], y[stop])],
                  base_margin_eval_set=None if bs is None else [bs], verbose=False)
        n = model.best_iteration + 1
        if margin_fn:
            bf, bt = margin_fn(full, [full, test])
        final = make_xgb(params, n, seed).fit(Xf.iloc[full], y[full], base_margin=bf)
        scores.append(float(average_precision_score(y[test], final.predict_proba(Xf.iloc[test], base_margin=bt)[:, 1])))
        trees.append(int(n))
    return scores, trees


def tune_xgb(Xf, y, month, grid: dict, n_iter: int, months=(7, 8, 9), margin_fn=None, seed: int = 42, verbose=True):
    """Random search over `grid`; each configuration is scored with rolling_eval (mean PR AUC)."""
    keys = list(grid)
    combos = list(itertools.product(*[grid[k] for k in keys]))
    picks = np.random.default_rng(seed).choice(len(combos), size=min(n_iter, len(combos)), replace=False)
    rows = []
    for i, ci in enumerate(picks):
        params = dict(zip(keys, combos[ci]))
        scores, trees = rolling_eval(Xf, y, month, params, months, margin_fn, seed=seed)
        rows.append({**params, "mean_pr_auc": float(np.mean(scores)), "fold_pr_auc": [round(v, 4) for v in scores], "trees_per_fold": trees})
        if verbose:
            print(f"[{i + 1:2d}/{len(picks)}] {params} -> folds {rows[-1]['fold_pr_auc']} mean {rows[-1]['mean_pr_auc']:.4f} trees {trees}", flush=True)
    return pd.DataFrame(rows).sort_values("mean_pr_auc", ascending=False).reset_index(drop=True)


class Platt:
    """Platt scaling: fit p_cal = sigmoid(a * logit(p) + b) on a held-out month."""

    def fit(self, p, y):
        self.m = LogisticRegression(C=1e6).fit(self._logit(p)[:, None], y)
        return self

    def transform(self, p):
        return self.m.predict_proba(self._logit(p)[:, None])[:, 1]

    @staticmethod
    def _logit(p):
        p = np.clip(np.asarray(p, float), 1e-6, 1 - 1e-6)
        return np.log(p / (1 - p))

Viva questions for you

Negative gradient of log loss?
r = y − p; each tree is fitted to it, so boosting is gradient descent in function space.
Learning rate vs number of trees?
Smaller steps need more trees and usually generalise better; early stopping chooses the count.
What was the early-stopping trap?
Stopping on the month you then report lets the stopping rule see the answers.
Why the median tree count?
One calm month (September) stops far too early; three folds give a steadier choice, made before testing.
Explain the hybrid.
XGBoost starts from logistic regression's logit and learns corrections; early stopping decides how many transfer.
Did boosting beat logistic regression?
No significant difference: the three learned models' 95% intervals overlap. All beat the majority class and the planner's rule.
Why did history features hurt boosting on October?
Trees split on absolute rate levels learned in a high-delay summer; autumn levels are lower (drift). A linear model only scales the rate.
Gain vs permutation importance?
Gain = loss reduction inside training; permutation = drop in held-out PR AUC when a feature is shuffled.
Why does the October threshold transfer badly?
The delay rate rose from 17.6% to 25.8%; a fixed score cut-off lands in a different place, while a calibrated-probability rule keeps precision near 0.5.

\begin{abstract}
We predict, at 05:00 on the day of a flight, whether a New York departure will arrive at least 15 minutes late, using only the published schedule, static aircraft data and the delays of earlier days. On 336,776 flights from 2013 we follow the full machine-learning life cycle: problem definition, data collection, EDA, preprocessing, a chronological split, leakage-safe feature engineering and selection, baselines, model selection, honest tuning of gradient boosting with rolling months, and a single evaluation on November--December (base rate 0.258). Tuned XGBoost, logistic regression and a hybrid that boosts from the logistic-regression prediction all beat a majority-class and a rule-of-thumb baseline, but their confidence intervals overlap: at this decision time gradient boosting does not significantly beat a regularised linear model. We show why: early stopping on the scored month inflates boosting's validation score, the useful tree count is unstable between months, and tree splits on absolute delay-rate levels suffer from seasonal drift.
\end{abstract}

\section{Introduction}
Late arrivals cascade through crews, gates and connections. A risk score available before the first departure lets planners add buffer or warn passengers. Following the course's five-step blueprint, the goal is to rank flights by delay risk (PR AUC) and to produce a flag list with precision $\geq 0.5$. The problem is binary classification: the U.S.\ BTS defines a flight as delayed when it arrives 15 or more minutes late, and delay minutes are heavily skewed (skewness 3.36). The project brief asks for gradient boosting; we test whether it is justified against simple baselines, as Phase 7 of the life cycle requires.

\section{Data and Exploratory Analysis}
The Kaggle \emph{Flights} (nycflights13) data contain 336,776 scheduled flights with a planes table. 8,255 cancelled and 1,175 diverted flights have no arrival delay and are left unlabelled (327,346 labelled flights). EDA on the training block only shows: aircraft data missing for 17.5\% of flights, concentrated in two carriers (systematic, so median imputation plus a missing indicator); 9.1\% IQR outliers in arrival delay, kept because they are real; a delay rate rising from 0.105 at 05:00 to 0.395 at 20:00 (propagation); and a moderate class imbalance, so no re-sampling. The daily delay rate has lag-1 autocorrelation 0.343 with a PACF cut-off after lag 1 (lag-2 PACF -0.109) and lag-7 autocorrelation 0.008, which motivates a 1-day history window and day-of-week features rather than weekly history.

\section{Method}
\subsection{Decision time and split}
Inputs must be known at 05:00 on day D. Post-departure columns are excluded and history windows end on day D$-$1; a unit test scrambles all outcomes from a cutoff date and checks that features on that date are unchanged. The split is chronological: train Jan 15--Aug 31 (205,642 flights, delay rate 0.2681), validation September (27,010, 0.1372), calibration October (28,618, 0.1756), test November--December (53,991, 0.2583), used once.

\subsection{Features and selection}
We build 41 numeric features in five groups (cyclic schedule time, calendar, aircraft, scheduled congestion, smoothed trailing delay history) plus carrier, origin and destination. History rates are shrunk towards the overall rate: $\hat r_g=(k_g+\alpha\,r_0)/(n_g+\alpha)$, $\alpha=10$. We tested filter (mutual information), VIF pruning (13 features have VIF $>10$), sequential forward selection, L1 and PCA (24 components for 95\% variance). SFS reached 0.279 PR AUC on September (all features: 0.261) but 0.281 on October (all features: 0.306): it over-fitted its selection month. No method beat all features on both held-out months, so all features were kept.

\subsection{Baselines and model selection}
Baselines: majority class, a planner's rule (the flight number's 28-day delay rate; September 0.190, October 0.269), and logistic regression, implemented from scratch with batch, mini-batch and stochastic gradient descent and checked against scikit-learn (C=0.001 chosen on September). On rolling months (train before July/August/September, test on that month) the mean PR AUC was 0.409 for logistic regression, 0.314 for a decision tree, 0.386 for a random forest and 0.366 for untuned XGBoost.

\subsection{Gradient boosting}
Boosting fits shallow trees to the negative gradient of log loss, $r=y-p$; our from-scratch version matches scikit-learn. Early-stopping on the month that is then scored is optimistic (mean 0.402 vs 0.379 for the same configuration), so we tune honestly: for month $m$, train on months before $m-1$, early-stop on $m-1$, refit on all months before $m$, and score $m$. A random search selected learning\_rate=0.05, max\_depth=3, min\_child\_weight=50, subsample=0.7, colsample\_bytree=0.8, reg\_lambda=10.0; the final tree count is the median over folds (74). A hybrid starts boosting from the logistic-regression logit (XGBoost \texttt{base\_margin}) so trees learn only corrections (median 8 trees). Rolling mean PR AUC: logistic regression 0.409, boosting 0.387, hybrid 0.405. Probabilities are Platt-calibrated on October.

\section{Conclusion}
At a 05:00 decision time, gradient boosting does not significantly outperform a regularised logistic regression; both clearly beat simple baselines. Honest tuning matters: a common early-stopping shortcut would have overstated boosting. Future work: drift-robust (relative) history features, rolling-origin model selection, information known at 05:00 such as weather forecasts, and temporal deep learning in Phase 2. \fillme{Phase 2 plan.}
